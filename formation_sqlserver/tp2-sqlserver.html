Transactions & Sécurité SQL Server

Gestion robuste des erreurs avec TRY...CATCH, XACT_ABORT et SAVEPOINTS. Protection de la base avec Logins, Users et Rôles.

1. Les Transactions T-SQL

SQL Server utilise BEGIN TRANSACTION (ou BEGIN TRAN) et la gestion d'erreurs est centralisée dans un bloc TRY...CATCH, similaire aux langages comme Java ou C#.

BEGIN TRY
    BEGIN TRANSACTION;
        UPDATE comptes SET solde = solde - 200 WHERE id = 1;
        UPDATE comptes SET solde = solde + 200 WHERE id = 2;
    COMMIT TRANSACTION;
    PRINT 'Virement réussi !';
END TRY
BEGIN CATCH
    IF @@TRANCOUNT > 0  -- Vérifie si une transaction est encore ouverte
        ROLLBACK TRANSACTION;
    PRINT 'Erreur: ' + ERROR_MESSAGE();
END CATCH;

SET XACT_ABORT ON

Une bonne pratique est d'activer SET XACT_ABORT ON au début de toute procédure transactionnelle. Cela garantit que si n'importe quelle instruction échoue, la transaction est automatiquement annulée.

SET XACT_ABORT ON;
BEGIN TRANSACTION;
    -- En cas d'erreur n'importe où ici, tout est automatiquement rollbacké
COMMIT;

Fonctions d'erreur dans le CATCH

  • ERROR_NUMBER() : Numéro de l'erreur SQL Server.
  • ERROR_MESSAGE() : Description de l'erreur.
  • ERROR_LINE() : Ligne où l'erreur s'est produite.
  • @@TRANCOUNT : Nombre de transactions ouvertes (0 = aucune).

2. Sécurité : Logins, Users et Rôles

-- Niveau Serveur : Créer un Login
CREATE LOGIN app_dev WITH PASSWORD = N'SecurePwd!1';

-- Niveau Base : Créer un User lié au Login
USE banking;
CREATE USER app_dev FOR LOGIN app_dev;

-- Assigner des rôles intégrés de la base (rapide)
ALTER ROLE db_datareader ADD MEMBER app_dev; -- Lecture seule sur toutes les tables
ALTER ROLE db_datawriter ADD MEMBER app_dev; -- Écriture sur toutes les tables

-- Créer un rôle personnalisé
CREATE ROLE conseiller;
GRANT SELECT ON comptes TO conseiller;
ALTER ROLE conseiller ADD MEMBER app_dev;

3. Base de travail — Système Bancaire

USE master;
IF EXISTS(SELECT * FROM sys.databases WHERE name='banking') DROP DATABASE banking;
CREATE DATABASE banking; GO
USE banking; GO

CREATE TABLE comptes (
    id INT IDENTITY(1,1) PRIMARY KEY,
    titulaire NVARCHAR(100) NOT NULL,
    solde DECIMAL(12,2) DEFAULT 0 CHECK (solde >= 0),
    type_compte NVARCHAR(20) DEFAULT N'courant'
);

CREATE TABLE virements (
    id INT IDENTITY(1,1) PRIMARY KEY,
    compte_source INT, compte_dest INT,
    montant DECIMAL(12,2),
    statut NVARCHAR(20) DEFAULT N'en_attente',
    effectue_le DATETIME2 DEFAULT GETDATE()
);

CREATE TABLE journal_erreurs (
    id INT IDENTITY(1,1) PRIMARY KEY,
    message NVARCHAR(MAX),
    contexte NVARCHAR(MAX),
    survenu_le DATETIME2 DEFAULT GETDATE()
);

4. Travaux Pratiques (10 exercices)

TP 01

Transaction simple avec TRY...CATCH

Effectuez un virement de 200€ d'Alice vers Bob dans un bloc TRY...CATCH. Vérifiez les soldes avant et après.

▶ Voir la solution 
SELECT id, titulaire, solde FROM comptes WHERE id IN (1,2);

BEGIN TRY
    BEGIN TRANSACTION;
        UPDATE comptes SET solde = solde - 200 WHERE id = 1;
        UPDATE comptes SET solde = solde + 200 WHERE id = 2;
    COMMIT TRANSACTION;
    PRINT 'Virement OK';
END TRY
BEGIN CATCH
    IF @@TRANCOUNT > 0 ROLLBACK;
    PRINT 'Erreur: ' + ERROR_MESSAGE();
END CATCH;

SELECT id, titulaire, solde FROM comptes WHERE id IN (1,2);
TP 02

Forcer un ROLLBACK avec THROW

Dans un bloc TRY...CATCH, débitez Alice de 100€ et levez manuellement une erreur avec THROW. Vérifiez que le CATCH rollback et que le solde est inchangé.

▶ Voir la solution 
SELECT solde FROM comptes WHERE id = 1;

BEGIN TRY
    BEGIN TRANSACTION;
        UPDATE comptes SET solde = solde - 100 WHERE id = 1;
        THROW 50000, 'Simulation d''erreur pour test', 1;
    COMMIT TRANSACTION;
END TRY
BEGIN CATCH
    IF @@TRANCOUNT > 0 ROLLBACK;
    PRINT 'Transaction annulée: ' + ERROR_MESSAGE();
END CATCH;

SELECT solde FROM comptes WHERE id = 1; -- Inchangé !
TP 03

Procédure : Virement sécurisé

Créez dbo.effectuer_virement @src, @dst, @mnt avec TRY...CATCH, vérification des fonds, insertion dans virements et journalisation des erreurs.

▶ Voir la solution 
CREATE OR ALTER PROCEDURE dbo.effectuer_virement
    @src INT, @dst INT, @mnt DECIMAL(12,2)
AS
BEGIN
    SET NOCOUNT ON; SET XACT_ABORT ON;
    BEGIN TRY
        IF @mnt <= 0 THROW 50001, 'Montant invalide', 1;
        DECLARE @solde DECIMAL(12,2);
        SELECT @solde = solde FROM comptes WITH (UPDLOCK) WHERE id = @src;
        IF @solde < @mnt THROW 50002, 'Fonds insuffisants', 1;
        BEGIN TRANSACTION;
            UPDATE comptes SET solde = solde - @mnt WHERE id = @src;
            UPDATE comptes SET solde = solde + @mnt WHERE id = @dst;
            INSERT INTO virements(compte_source, compte_dest, montant, statut) VALUES(@src,@dst,@mnt,N'effectue');
        COMMIT;
        PRINT N'Virement de ' + CAST(@mnt AS NVARCHAR) + N'€ OK';
    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0 ROLLBACK;
        INSERT INTO journal_erreurs(message, contexte) VALUES(ERROR_MESSAGE(), N'Virement ' + CAST(@src AS NVARCHAR) + N' → ' + CAST(@dst AS NVARCHAR));
        THROW;
    END CATCH;
END;
GO

EXEC dbo.effectuer_virement 1, 2, 50;
EXEC dbo.effectuer_virement 5, 1, 100; -- Doit échouer
SELECT * FROM virements;
TP 04

SAVEPOINT (SQL Server)

SQL Server utilise SAVE TRANSACTION point_name. Effectuez 2 virements, créez un point de sauvegarde, tentez un 3ème qui échoue et revenez au savepoint.

▶ Voir la solution 
BEGIN TRANSACTION;
    UPDATE comptes SET solde = solde - 30 WHERE id = 1;
    UPDATE comptes SET solde = solde + 30 WHERE id = 2;
    SAVE TRANSACTION apres_deux;  -- Point de sauvegarde

    BEGIN TRY
        UPDATE comptes SET solde = solde - 500 WHERE id = 5;
        -- La contrainte CHECK va déclencher une erreur (solde >= 0)
    END TRY
    BEGIN CATCH
        ROLLBACK TRANSACTION apres_deux;  -- Retour au savepoint, pas au tout début
        PRINT 'Virement 3 annulé: ' + ERROR_MESSAGE();
    END CATCH;
COMMIT;

SELECT id, titulaire, solde FROM comptes ORDER BY id;

TP 07

Trigger : Bloquer les virements suspects

Créez un trigger AFTER INSERT sur virements qui rollback la transaction si le montant dépasse 10 000€.

▶ Voir la solution 
CREATE OR ALTER TRIGGER trg_virement_suspect
ON virements
AFTER INSERT
AS
BEGIN
    SET NOCOUNT ON;
    IF EXISTS (SELECT 1 FROM inserted WHERE montant > 10000)
    BEGIN
        INSERT INTO journal_erreurs(message) SELECT N'Virement bloqué: ' + CAST(montant AS NVARCHAR) + N'€' FROM inserted;
        ROLLBACK TRANSACTION;
        THROW 50003, 'Virement refusé: dépasse le plafond de 10 000€', 1;
    END;
END;
GO

INSERT INTO virements(compte_source, compte_dest, montant) VALUES(4, 1, 15000);
TP 08

Observer l'isolation (NOLOCK hint)

SQL Server permet de lire des données non-commitées avec l'hint WITH (NOLOCK) — c'est un "dirty read". Démontrez la différence entre une lecture normale et une lecture avec NOLOCK pendant une transaction en cours.

▶ Voir la solution 
-- SESSION A : Transaction non commitée
BEGIN TRANSACTION;
UPDATE comptes SET solde = 99999 WHERE id = 1;
-- NE PAS COMMITTER

-- SESSION B : Lecture normale (va attendre le verrou)
SELECT solde FROM comptes WHERE id = 1;  -- BLOQUE (attend la session A)

-- SESSION B : Lecture avec NOLOCK (lit la donnée non-commitée = dirty read)
SELECT solde FROM comptes WITH (NOLOCK) WHERE id = 1;  -- Retourne 99999€ !
-- ATTENTION : Éviter NOLOCK en production sauf cas très spécifiques

-- SESSION A : Rollback
ROLLBACK;
SELECT solde FROM comptes WHERE id = 1;  -- Retourne la vraie valeur
TP 09

Vue sécurisée (VIEW) avec EXECUTE AS

Créez une vue v_comptes_masques et spécifiez WITH SCHEMABINDING pour l'empêcher de voir les données si on modifie les tables sous-jacentes.

▶ Voir la solution 
CREATE OR ALTER VIEW dbo.v_comptes_masques
WITH SCHEMABINDING  -- Empêche de modifier les tables sans supprimer la vue
AS
SELECT
    id,
    titulaire,
    type_compte,
    CASE
        WHEN solde > 1000 THEN N'Solide'
        WHEN solde > 0 THEN N'Faible'
        ELSE N'Insuffisant'
    END AS niveau_solde
FROM dbo.comptes;
GO

GRANT SELECT ON dbo.v_comptes_masques TO conseiller_martin;
SELECT * FROM dbo.v_comptes_masques;
TP 10

Procédure de clôture mensuelle

Créez dbo.cloture_mensuelle dans un TRY...CATCH : 2% sur épargne, frais -5€ sur courants, rapport dans journal_erreurs.

▶ Voir la solution 
CREATE OR ALTER PROCEDURE dbo.cloture_mensuelle
AS
BEGIN
    SET NOCOUNT ON;
    DECLARE @nb_epargne INT, @nb_courant INT;
    BEGIN TRY
        BEGIN TRANSACTION;
            UPDATE comptes SET solde = ROUND(solde * 1.02, 2) WHERE type_compte = N'epargne';
            SET @nb_epargne = @@ROWCOUNT;
            UPDATE comptes SET solde = solde - 5 WHERE type_compte = N'courant' AND solde > 0;
            SET @nb_courant = @@ROWCOUNT;
            INSERT INTO journal_erreurs(message, contexte)
            VALUES(N'CLOTURE MENSUELLE OK', CAST(@nb_epargne AS NVARCHAR) + N' épargne, ' + CAST(@nb_courant AS NVARCHAR) + N' courants');
        COMMIT;
        PRINT N'Clôture OK: ' + CAST(@nb_epargne AS NVARCHAR) + N' épargne, ' + CAST(@nb_courant AS NVARCHAR) + N' courants';
    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0 ROLLBACK;
        THROW;
    END CATCH;
END;
GO

EXEC dbo.cloture_mensuelle;
SELECT id, titulaire, solde, type_compte FROM comptes ORDER BY type_compte;
