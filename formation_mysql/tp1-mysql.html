Routines & Triggers MySQL

Maîtrisez la programmation côté serveur avec MySQL. La syntaxe DELIMITER est particulière mais puissante — soyons méthodiques.

1. Pourquoi le DELIMITER dans MySQL ?

MySQL utilise le point-virgule ; pour savoir quand une commande se termine. Mais à l'intérieur d'une procédure, on a plusieurs ;. Il faut donc changer temporairement le séparateur pour que MySQL ne s'arrête pas au premier ; rencontré.

DELIMITER //   -- On dit à MySQL "maintenant la commande se termine par //"

CREATE PROCEDURE ma_procedure()
BEGIN
    SELECT 'Bonjour' ;  -- Ce ";" est ignoré par MySQL, c'est celui de la procédure
    SELECT 'Monde' ;    -- Idem
END //          -- MySQL comprend que la procédure est finie ici

DELIMITER ;     -- On remet le séparateur normal

Structure d'une procédure stockée

DELIMITER //
CREATE PROCEDURE nom_procedure(
    IN param1 INT,       -- Paramètre en entrée
    OUT param2 VARCHAR(100), -- Paramètre en sortie
    INOUT param3 DECIMAL(10,2) -- Entrée ET sortie
)
BEGIN
    DECLARE ma_variable INT DEFAULT 0;  -- Variables locales

    IF param1 > 10 THEN
        SET param2 = 'Grand';
    ELSE
        SET param2 = 'Petit';
    END IF;
END //
DELIMITER ;

Variables spéciales dans un Trigger MySQL

  • NEW.colonne — La nouvelle valeur (INSERT / UPDATE).
  • OLD.colonne — L'ancienne valeur (UPDATE / DELETE).
  • Les triggers MySQL s'exécutent FOR EACH ROW (ligne par ligne) — pas de trigger Statement-Level.
Note importante : Dans MySQL, un trigger ne peut pas appeler de procédures qui contiennent des transactions (COMMIT/ROLLBACK). La logique des triggers doit rester simple.

CREATE DATABASE IF NOT EXISTS ecommerce CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE ecommerce;

DROP TABLE IF EXISTS commandes, produits, clients, audit_prix;

CREATE TABLE clients (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nom VARCHAR(100) NOT NULL,
    email VARCHAR(150) UNIQUE,
    points_fidelite INT DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB;

CREATE TABLE produits (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nom VARCHAR(100) NOT NULL,
    prix DECIMAL(10,2) CHECK (prix >= 0),
    stock INT DEFAULT 0,
    categorie VARCHAR(50)
) ENGINE=InnoDB;

CREATE TABLE commandes (
    id INT AUTO_INCREMENT PRIMARY KEY,
    client_id INT, produit_id INT,
    quantite INT CHECK (quantite > 0),
    prix_total DECIMAL(10,2),
    date_cmd TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (client_id) REFERENCES clients(id) ON DELETE CASCADE,
    FOREIGN KEY (produit_id) REFERENCES produits(id)
) ENGINE=InnoDB;

3. Travaux Pratiques (10 exercices)

TP 01

Fonction : Calcul de la remise fidélité

Créez une fonction get_remise(p_id INT) qui retourne le taux de remise (DECIMAL). Règle : < 100 pts → 0%, 100 à 499 → 0.10, ≥ 500 → 0.20.

▶ Voir la solution 
DELIMITER //
CREATE FUNCTION get_remise(p_id INT)
RETURNS DECIMAL(4,2)
READS SQL DATA
DETERMINISTIC
BEGIN
    DECLARE pts INT;
    SELECT points_fidelite INTO pts FROM clients WHERE id = p_id;
    IF pts IS NULL THEN RETURN 0.00; END IF;
    IF pts >= 500 THEN RETURN 0.20;
    ELSEIF pts >= 100 THEN RETURN 0.10;
    ELSE RETURN 0.00;
    END IF;
END //
DELIMITER ;

-- Test :
SELECT nom, points_fidelite, get_remise(id)*100 AS remise_pct FROM clients;
TP 02

Fonction : Prix final avec remise

Créez une fonction prix_final(p_client INT, p_produit INT) qui appelle get_remise et retourne le prix après réduction.

▶ Voir la solution 
DELIMITER //
CREATE FUNCTION prix_final(p_client INT, p_produit INT)
RETURNS DECIMAL(10,2)
READS SQL DATA
DETERMINISTIC
BEGIN
    DECLARE v_prix DECIMAL(10,2);
    DECLARE v_remise DECIMAL(4,2);
    SELECT prix INTO v_prix FROM produits WHERE id = p_produit;
    SET v_remise = get_remise(p_client);
    RETURN ROUND(v_prix * (1 - v_remise), 2);
END //
DELIMITER ;

SELECT prix_final(3, 1); -- Clara (500pts) sur Laptop (1200€) → 960€
TP 03

Procédure : Passer une commande

Créez une procédure passer_commande(p_client, p_produit, p_qte) qui vérifie le stock, insère la commande avec le prix final et déduit le stock. Utilisez SIGNAL SQLSTATE pour lever l'erreur.

▶ Voir la solution 
DELIMITER //
CREATE PROCEDURE passer_commande(IN p_client INT, IN p_produit INT, IN p_qte INT)
BEGIN
    DECLARE v_stock INT;
    DECLARE v_prix DECIMAL(10,2);

    SELECT stock INTO v_stock FROM produits WHERE id = p_produit;

    IF v_stock < p_qte THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Stock insuffisant pour cette commande.';
    END IF;

    SET v_prix = prix_final(p_client, p_produit) * p_qte;

    INSERT INTO commandes (client_id, produit_id, quantite, prix_total)
    VALUES (p_client, p_produit, p_qte, v_prix);

    UPDATE produits SET stock = stock - p_qte WHERE id = p_produit;

    SELECT CONCAT('Commande validée. Total: ', v_prix, ' EUR') AS message;
END //
DELIMITER ;

-- Test :
CALL passer_commande(1, 2, 3);
SELECT * FROM commandes;
SELECT id, nom, stock FROM produits;

Trigger : Bloquer le stock négatif

DELIMITER //
CREATE TRIGGER trg_stock_negatif
BEFORE UPDATE ON produits
FOR EACH ROW
BEGIN
    IF NEW.stock < 0 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Le stock ne peut pas être négatif.';
    END IF;
END //
DELIMITER ;

CREATE TABLE audit_prix (

CREATE TRIGGER trg_audit_prix
AFTER UPDATE ON produits
FOR EACH ROW
BEGIN
    IF NEW.prix <> OLD.prix THEN
        INSERT INTO audit_prix(produit_id, ancien_prix, nouveau_prix)
        VALUES (OLD.id, OLD.prix, NEW.prix);

END //

TP 06

Fonction : Valeur totale du stock par catégorie

Créez une fonction valeur_stock_categorie(cat VARCHAR(50)) qui retourne la valeur totale (prix × stock) pour une catégorie.

▶ Voir la solution 
DELIMITER //
CREATE FUNCTION valeur_stock_categorie(p_cat VARCHAR(50))
RETURNS DECIMAL(12,2)
READS SQL DATA
DETERMINISTIC
BEGIN
    DECLARE v_total DECIMAL(12,2);
    SELECT COALESCE(SUM(prix * stock), 0) INTO v_total
    FROM produits WHERE categorie = p_cat;
    RETURN v_total;
END //
DELIMITER ;

SELECT valeur_stock_categorie('Informatique');
SELECT valeur_stock_categorie('Accessoires');
TP 07

Trigger : Accumulation automatique des points

Créez un trigger AFTER INSERT sur commandes qui ajoute 1 point de fidélité par euro dépensé au client.

▶ Voir la solution 
DELIMITER //
CREATE TRIGGER trg_points_fidelite
AFTER INSERT ON commandes
FOR EACH ROW
BEGIN
    UPDATE clients
    SET points_fidelite = points_fidelite + ROUND(NEW.prix_total)
    WHERE id = NEW.client_id;
END //
DELIMITER ;

TP 08

Procédure : Réapprovisionner le stock

Créez une procédure reapprovisionner(p_produit INT, p_qte INT) qui rejette les quantités nulles ou négatives et affiche un message de confirmation.

▶ Voir la solution 
DELIMITER //
CREATE PROCEDURE reapprovisionner(IN p_produit INT, IN p_qte INT)
BEGIN
    DECLARE v_nom VARCHAR(100);
    DECLARE v_stock_avant INT;

    IF p_qte <= 0 THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'La quantité doit être positive.';
    END IF;

    SELECT nom, stock INTO v_nom, v_stock_avant FROM produits WHERE id = p_produit;
    UPDATE produits SET stock = stock + p_qte WHERE id = p_produit;
    SELECT CONCAT('Réappro OK: ', v_nom, ' — ', v_stock_avant, ' → ', v_stock_avant + p_qte, ' unités') AS resultat;
END //
DELIMITER ;

CALL reapprovisionner(4, 10);
CALL reapprovisionner(1, -5); -- Doit échouer
TP 09

Procédure avec curseur : Historique client

Créez une procédure afficher_historique(p_id INT) qui utilise un curseur pour parcourir les commandes d'un client et les afficher une par une avec le nom du produit.

▶ Voir la solution 
DELIMITER //
CREATE PROCEDURE afficher_historique(IN p_id INT)
BEGIN
    DECLARE done INT DEFAULT 0;
    DECLARE v_produit VARCHAR(100);
    DECLARE v_qte INT;
    DECLARE v_total DECIMAL(10,2);
    DECLARE v_date TIMESTAMP;

    DECLARE cur CURSOR FOR
        SELECT p.nom, c.quantite, c.prix_total, c.date_cmd
        FROM commandes c JOIN produits p ON p.id = c.produit_id
        WHERE c.client_id = p_id ORDER BY c.date_cmd DESC;

    DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = 1;

    OPEN cur;
    read_loop: LOOP
        FETCH cur INTO v_produit, v_qte, v_total, v_date;
        IF done THEN LEAVE read_loop; END IF;
        SELECT v_produit AS produit, v_qte AS quantite, v_total AS total, v_date AS date_achat;
    END LOOP;
    CLOSE cur;
END //
DELIMITER ;

CALL afficher_historique(1);
TP 10

Trigger : Empêcher la suppression de commandes récentes

Créez une table tentatives_suppression et un trigger BEFORE DELETE sur commandes qui bloque la suppression de commandes de moins de 30 jours.

▶ Voir la solution 
CREATE TABLE tentatives_suppression (
    id INT AUTO_INCREMENT PRIMARY KEY,
    commande_id INT,
    tente_le TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

DELIMITER //
CREATE TRIGGER trg_protect_commandes
BEFORE DELETE ON commandes
FOR EACH ROW
BEGIN
    IF OLD.date_cmd > DATE_SUB(NOW(), INTERVAL 30 DAY) THEN
        INSERT INTO tentatives_suppression(commande_id) VALUES (OLD.id);
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Suppression refusée : commande de moins de 30 jours.';
    END IF;
END //
DELIMITER ;

DELETE FROM commandes WHERE id = 1;
SELECT * FROM tentatives_suppression;
