PL/pgSQL : Fonctions, Procédures et Triggers

Programmez directement dans PostgreSQL. Apprenez à encapsuler la logique métier côté serveur pour des bases de données plus robustes, sécurisées et performantes.

1. Pourquoi programmer côté base de données ?

En architecture classique, la logique métier est dans l'application (PHP, Java, Python). Mais certaines opérations critiques (intégrité des données, calculs intensifs, audit) sont bien plus efficaces et sûres si elles vivent dans le moteur de base de données lui-même.

  • Performance : Les données ne transitent pas inutilement par le réseau.
  • Intégrité garantie : Les règles s'appliquent même si on accède à la BDD via un autre outil (Excel, DBeaver...).
  • Centralisation : La logique est en un seul endroit, peu importe le langage de l'application.

Structure de base d'une fonction PL/pgSQL

CREATE OR REPLACE FUNCTION nom_fonction(param1 TYPE, param2 TYPE)
RETURNS TYPE_DE_RETOUR AS $$
DECLARE
    -- Déclaration des variables locales
    ma_variable INT := 0;
BEGIN
    -- Corps de la fonction (logique SQL + impératif)
    -- ...
    RETURN ma_variable;
END;
$$ LANGUAGE plpgsql;

Les différences clés

FUNCTION vs PROCEDURE vs TRIGGER :
— Fonction : retourne une valeur, utilisable dans un SELECT.
— Procédure (PG ≥11) : pas de retour, peut gérer ses propres COMMIT/ROLLBACK.
— Trigger : une FONCTION spéciale (retournant TRIGGER) exécutée automatiquement sur un événement INSERT/UPDATE/DELETE.

Variables spéciales dans un Trigger

  • NEW — La nouvelle ligne (disponible sur INSERT et UPDATE).
  • OLD — L'ancienne ligne (disponible sur UPDATE et DELETE).
  • TG_OP — L'opération déclenchante ('INSERT', 'UPDATE', 'DELETE').
  • TG_TABLE_NAME — Le nom de la table concernée.

Exemple complet : Trigger d'audit

-- 1. Table d'audit
CREATE TABLE audit_log (
    id SERIAL PRIMARY KEY,
    table_name TEXT,
    operation TEXT,
    changed_by TEXT DEFAULT current_user,
    changed_at TIMESTAMP DEFAULT now()
);

-- 2. Fonction trigger
CREATE OR REPLACE FUNCTION log_operation()
RETURNS TRIGGER AS $$
BEGIN
    INSERT INTO audit_log(table_name, operation)
    VALUES (TG_TABLE_NAME, TG_OP);
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- 3. Attacher le trigger
CREATE TRIGGER trg_audit
AFTER INSERT OR UPDATE OR DELETE ON clients
FOR EACH ROW EXECUTE FUNCTION log_operation();

3. Travaux Pratiques (10 exercices)

TP 01

Fonction : Calcul de la remise fidélité

Créez une fonction get_remise(client_id INT) qui retourne un taux de remise DECIMAL. Règle : < 100 pts → 0%, entre 100 et 499 → 10%, ≥ 500 → 20%.

▶ Voir la solution 
CREATE OR REPLACE FUNCTION get_remise(p_id INT)
RETURNS DECIMAL AS $$
DECLARE
    pts INT;
BEGIN
    SELECT points_fidelite INTO pts FROM clients WHERE id = p_id;
    IF pts IS NULL THEN RETURN 0; END IF;
    IF pts >= 500 THEN RETURN 0.20;
    ELSIF pts >= 100 THEN RETURN 0.10;
    ELSE RETURN 0.00;
    END IF;
END;
$$ LANGUAGE plpgsql;

-- Test :
SELECT nom, points_fidelite, get_remise(id)*100 || '%' AS remise FROM clients;
TP 02

Fonction : Prix final avec remise

Créez une fonction prix_final(p_client INT, p_produit INT) qui retourne le prix d'un produit après application de la remise du client.

▶ Voir la solution 
CREATE OR REPLACE FUNCTION prix_final(p_client INT, p_produit INT)
RETURNS DECIMAL AS $$
DECLARE
    v_prix DECIMAL;
    v_remise DECIMAL;
BEGIN
    SELECT prix INTO v_prix FROM produits WHERE id = p_produit;
    v_remise := get_remise(p_client);
    RETURN ROUND(v_prix * (1 - v_remise), 2);
END;
$$ LANGUAGE plpgsql;

-- Test : Prix du Laptop (1200€) pour Alice (20%) → 960€
SELECT prix_final(3, 1); -- Clara (500pts) sur Laptop
TP 03

Procédure : Passer une commande

Créez une procédure passer_commande(p_client, p_produit, p_qte) qui : (1) vérifie le stock, (2) lève une exception si insuffisant, (3) insère la commande avec le prix final, (4) déduit le stock.

▶ Voir la solution 
CREATE OR REPLACE PROCEDURE passer_commande(
    p_client INT, p_produit INT, p_qte INT
) AS $$
DECLARE
    v_stock INT;
    v_prix_final DECIMAL;
BEGIN
    SELECT stock INTO v_stock FROM produits WHERE id = p_produit;

    IF v_stock < p_qte THEN
        RAISE EXCEPTION 'Stock insuffisant: % disponible(s), % demandé(s)', v_stock, p_qte;
    END IF;

    v_prix_final := prix_final(p_client, p_produit) * p_qte;

    INSERT INTO commandes (client_id, produit_id, quantite, prix_total)
    VALUES (p_client, p_produit, p_qte, v_prix_final);

    UPDATE produits SET stock = stock - p_qte WHERE id = p_produit;

    RAISE NOTICE 'Commande validée. Total: % EUR', v_prix_final;
END;
$$ LANGUAGE plpgsql;

-- Test :
CALL passer_commande(1, 2, 3);  -- Alice commande 3 souris
SELECT * FROM commandes;
SELECT id, nom, stock FROM produits;
TP 04

Trigger : Bloquer le stock négatif

Créez un trigger BEFORE UPDATE sur produits qui empêche toute modification portant le stock en dessous de 0.

▶ Voir la solution 
CREATE OR REPLACE FUNCTION check_stock_negatif()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.stock < 0 THEN
        RAISE EXCEPTION 'Opération refusée: le stock de "%" ne peut pas être négatif (valeur: %)', NEW.nom, NEW.stock;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_stock_negatif
BEFORE UPDATE ON produits
FOR EACH ROW
WHEN (NEW.stock IS DISTINCT FROM OLD.stock)
EXECUTE FUNCTION check_stock_negatif();

-- Test : doit échouer
UPDATE produits SET stock = stock - 100 WHERE id = 2;
TP 05

Trigger : Audit des changements de prix

Créez une table audit_prix(id, produit_id, ancien_prix, nouveau_prix, modifie_par, modifie_le) et un trigger qui l'alimente automatiquement à chaque modification du prix d'un produit.

▶ Voir la solution 
CREATE TABLE audit_prix (
    id SERIAL PRIMARY KEY,
    produit_id INT,
    ancien_prix DECIMAL(10,2),
    nouveau_prix DECIMAL(10,2),
    modifie_par TEXT DEFAULT current_user,
    modifie_le TIMESTAMP DEFAULT now()
);

CREATE OR REPLACE FUNCTION log_changement_prix()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.prix <> OLD.prix THEN
        INSERT INTO audit_prix(produit_id, ancien_prix, nouveau_prix)
        VALUES (OLD.id, OLD.prix, NEW.prix);
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_audit_prix
AFTER UPDATE ON produits
FOR EACH ROW EXECUTE FUNCTION log_changement_prix();

-- Test :
UPDATE produits SET prix = 1350.00 WHERE id = 1;
UPDATE produits SET prix = 40.00 WHERE id = 2;
SELECT * FROM audit_prix;
TP 06

Fonction : Valeur totale du stock

Créez une fonction valeur_stock_categorie(cat TEXT) qui retourne la valeur totale du stock (prix × quantité) pour une catégorie donnée.

▶ Voir la solution 
CREATE OR REPLACE FUNCTION valeur_stock_categorie(p_cat TEXT)
RETURNS DECIMAL AS $$
DECLARE
    v_total DECIMAL;
BEGIN
    SELECT COALESCE(SUM(prix * stock), 0)
    INTO v_total
    FROM produits
    WHERE categorie = p_cat;
    RETURN v_total;
END;
$$ LANGUAGE plpgsql;

-- Test :
SELECT valeur_stock_categorie('Informatique');
SELECT valeur_stock_categorie('Accessoires');
TP 07

Trigger : Accumulation automatique des points

Créez un trigger AFTER INSERT sur commandes qui ajoute 1 point de fidélité par euro dépensé (arrondi) au client concerné.

▶ Voir la solution 
CREATE OR REPLACE FUNCTION ajouter_points_fidelite()
RETURNS TRIGGER AS $$
BEGIN
    UPDATE clients
    SET points_fidelite = points_fidelite + ROUND(NEW.prix_total)
    WHERE id = NEW.client_id;

    RAISE NOTICE 'Client % a gagné % points', NEW.client_id, ROUND(NEW.prix_total);
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_points_fidelite
AFTER INSERT ON commandes
FOR EACH ROW EXECUTE FUNCTION ajouter_points_fidelite();

-- Test :
CALL passer_commande(2, 3, 1);  -- Bob commande 1 clavier (95€)
SELECT id, nom, points_fidelite FROM clients WHERE id = 2;
TP 08

Procédure : Réapprovisionner le stock

Créez une procédure reapprovisionner(p_produit INT, p_quantite INT) qui incrémente le stock. Elle doit rejeter toute quantité nulle ou négative et afficher un message de confirmation.

▶ Voir la solution 
CREATE OR REPLACE PROCEDURE reapprovisionner(p_produit INT, p_qte INT)
AS $$
DECLARE
    v_nom TEXT;
    v_stock_avant INT;
BEGIN
    IF p_qte <= 0 THEN
        RAISE EXCEPTION 'La quantité doit être positive (reçu: %)', p_qte;
    END IF;

    SELECT nom, stock INTO v_nom, v_stock_avant FROM produits WHERE id = p_produit;

    UPDATE produits SET stock = stock + p_qte WHERE id = p_produit;

    RAISE NOTICE 'Réappro OK: "%" — % → % unités', v_nom, v_stock_avant, v_stock_avant + p_qte;
END;
$$ LANGUAGE plpgsql;

-- Tests :
CALL reapprovisionner(4, 10);  -- OK
CALL reapprovisionner(1, -5);  -- Doit échouer
TP 09

Fonction retournant une table : Historique client

Créez une fonction historique_client(p_id INT) qui retourne un tableau (RETURNS TABLE) avec : nom_produit, quantite, prix_total, date de chaque commande d'un client.

▶ Voir la solution 
CREATE OR REPLACE FUNCTION historique_client(p_id INT)
RETURNS TABLE(
    produit TEXT,
    qte INT,
    total DECIMAL,
    date_achat TIMESTAMP
) AS $$
BEGIN
    RETURN QUERY
    SELECT
        p.nom,
        c.quantite,
        c.prix_total,
        c.date_cmd
    FROM commandes c
    JOIN produits p ON p.id = c.produit_id
    WHERE c.client_id = p_id
    ORDER BY c.date_cmd DESC;
END;
$$ LANGUAGE plpgsql;

-- Test :
SELECT * FROM historique_client(1);  -- Historique d'Alice
TP 10

Trigger : Empêcher la suppression des commandes

Créez un trigger BEFORE DELETE sur commandes qui bloque toute suppression d'une commande de moins de 30 jours et enregistre la tentative dans une table tentatives_suppression.

▶ Voir la solution 
CREATE TABLE tentatives_suppression (
    id SERIAL PRIMARY KEY,
    commande_id INT,
    tente_par TEXT DEFAULT current_user,
    tente_le TIMESTAMP DEFAULT now()
);

CREATE OR REPLACE FUNCTION bloquer_suppression_commande()
RETURNS TRIGGER AS $$
BEGIN
    IF OLD.date_cmd > now() - INTERVAL '30 days' THEN
        INSERT INTO tentatives_suppression(commande_id) VALUES (OLD.id);
        RAISE EXCEPTION 'Suppression refusée : commande % datée du % (moins de 30 jours)',
            OLD.id, OLD.date_cmd::date;
    END IF;
    RETURN OLD;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_protect_commandes
BEFORE DELETE ON commandes
FOR EACH ROW EXECUTE FUNCTION bloquer_suppression_commande();

-- Test (doit échouer pour les commandes récentes) :
DELETE FROM commandes WHERE id = 1;
SELECT * FROM tentatives_suppression;
